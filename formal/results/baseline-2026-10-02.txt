  "∀ C S sid k1 k2 #i.

"∃ C S sid k1 k2 #i.
  (Commit( C, S, <'server', sid, k1, k2> ) @ #i) ∧

  "∀ C S sid k1 k2 #i.
    (((Commit( C, S, <'server', sid, k1, k2> ) @ #i) ∧ (Honest( S ) @ #i)) ∧

    (∃ #j. Commit( S, C, <'client', sid, k1, k2> ) @ #j)"

"∃ C S sid k1 k2 #i.
  (Commit( C, S, <'server', sid, k1, k2> ) @ #i) ∧ (Honest( S ) @ #i)
